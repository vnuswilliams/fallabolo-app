matchrh-mcp.ts
// Connexion MCP MatchRH
import { MatchRH } from 'matchrh-mcp'
 
// Recruteur — top candidats
const top = await MatchRH.getTopCandidates({
  offreId: 'comptable-douala-01',
  limit: 5
})
// [{ name: 'Sylvie', score: 91% },
// { name: 'Patrick', score: 87% }]
 
// Candidat — offres compatibles
const offres = await MatchRH.getMatchingOffers({
  candidatId: 'mon-profil',
  minScore: 75
})
// [{ titre: 'Chef Comptable', score: 88% },
// { titre: 'Auditeur', score: 82% }]
